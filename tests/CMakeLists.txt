function(er_test name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE editroom_editor Qt6::Test)
  target_compile_definitions(${name} PRIVATE ER_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()
